```ts
import {
  Entity, PrimaryGeneratedColumn, Column, CreateDateColumn, UpdateDateColumn, DeleteDateColumn, Index,
} from 'typeorm';
import { QuizQuestionKind } from '@constants/enums';
import { COMMON_SCOPE } from '@shared/content-scope';

export interface QuizOption {
  text: string;
  answer?: boolean;   // knowledge 用：正确选项标记
  score?: number;     // assessment 用：选项分值
  goto?: string;      // assessment 用：跳转目标题目 code（覆盖卷内顺序）
  explain?: string;   // knowledge 用：解析（不下发，答后返回）
}

@Entity('quiz_questions')
@Index('uq_quiz_question_code', ['appScope', 'code'], { unique: true })
export class QuizQuestion {
  @PrimaryGeneratedColumn({ type: 'bigint' })
  id: string;

  @Column({ type: 'varchar', length: 64 })
  code: string;

  @Column({ type: 'enum', enum: QuizQuestionKind })
  kind: QuizQuestionKind;

  @Column({ type: 'text' })
  content: string;

  @Column({ type: 'jsonb', default: '[]' })
  options: QuizOption[];

  @Column({ name: 'multi_select', type: 'boolean', default: false })
  multiSelect: boolean;

  @Column({ type: 'varchar', length: 64, nullable: true })
  category: string | null;

  @Column({ type: 'int', default: 1 })
  difficulty: number;

  @Column({ type: 'jsonb', default: '[]' })
  tags: string[];

  /** { currencyType: 'GOLD', amount: 10 } —— 知识题答对（首次）发放 */
  @Column({ name: 'reward_json', type: 'jsonb', default: '{}' })
  rewardJson: Record<string, any>;

  @Column({ name: 'app_scope', type: 'varchar', length: 32, default: COMMON_SCOPE })
  appScope: string;

  @CreateDateColumn({ name: 'created_at' }) createdAt: Date;
  @UpdateDateColumn({ name: 'updated_at' }) updatedAt: Date;
  @DeleteDateColumn({ name: 'deleted_at' }) deletedAt: Date | null;
}
```

- [ ] **Step 3.1: 写失败测试（quiz-flow.spec.ts，纯函数直测）**

构造用例覆盖：① goto 指向卷外 code → 死链错误；② A→B→A 环 → 死循环错误；③ 分支节点某选项跳进的子图无终点 → 不可达终点错误；④ 合法卷（默认顺序 + goto 跳过中段 + 尾题收尾）→ 无错误；⑤ 卷内含 kind=knowledge 题 → 错误；⑥ question.appScope 与卷不同且非 common → 错误；⑦ scoring_rule 引用不存在的 result code → 错误；⑧ dimension 模式缺 dim 名 → 错误。

校验器输入输出签名：

```ts
export interface FlowItemInput {
  questionId: string;
  sortOrder: number;
  nextQuestionId: string | null;
  dimension: string | null;
  question: { id: string; code: string; kind: string; appScope: string; options: Array<{ goto?: string }> };
}
export interface ScoringRuleInput {
  mode: 'total' | 'dimension';
  results?: Array<{ min: number; max: number; result: string }>;
  dims?: Array<{ name: string; results: Array<{ min: number; max: number; result: string }> }>;
}
/** 返回错误列表；空数组 = 通过 */
export function validateAssessmentFlow(input: {
  items: FlowItemInput[];
  startQuestionId: string | null;
  scoringRule: ScoringRuleInput;
  assessmentScope: string;
  knownResultCodes: Set<string>;
}): string[];
```

规则：跳转解析优先级 goto > nextQuestionId > 默认顺序；每个选项恰好一条出边（option.goto 存在用之，否则 item.nextQuestionId，否则按 sort_order 下一题，末题出度为 0 = 终点）；从起点可达子图内**不允许任何环**且**每个节点必须能到达出度为 0 的终点**；result code 必须在 `knownResultCodes` 且其 scope = common 或 = assessmentScope。

```ts

@Post('api/client/v1/quiz/assessments/:code/start')   // 开始测评 → { sessionId, question }
@Post('api/client/v1/quiz/sessions/:id/answer')       // 答题 → { finished, question? } | { finished, result, reward? }
```

- `start(playerId, code, appCode = null)`：`assessmentRepo.findOne({ where: { code, status: PUBLISHED, ...visibleTo(appCode) } })` → 无则 46001；items 按 sortOrder 排序；首题 = startQuestionId ?? 首个 item；建 session（app_code 存 `appCode ?? DEFAULT_APP_CODE`）→ 返回 `{ sessionId, question: sanitizeQuestion(q) }`
- `answer(playerId, sessionId, questionId, selected)`：会话校验链 46002（id+playerId 查不到）→ 46008（status=finished）→ 46003（startedAt < now-24h，**未完成才判过期**）→ 46004（currentQuestionId 不匹配）→ 46005（selected 空或含越界序号）
- 计分：`total_score += Σ option.score ?? 0`；item.dimension 非空时 `dim_scores[dim] += score`
- 跳转：第一个选中选项的 `goto`（题目 code → 本卷 items 内反查 id）> item.nextQuestionId > sort_order 下一题；无下一题 → finished
- 结算：`resolveResult(scoringRule, total, dimScores, appCode)`——total 模式找 `min<=total<=max`；dimension 模式逐 dim 找区间；result 按 code 查（`visibleTo(appCode)`）；无命中 → 46007（日志 warn，不抛给玩家的场景不存在——admin 已校验，防御性保留）

- 返回 `{ finished: true, result: { code, title, content }, reward?: {...} }` 或 `{ finished: false, question }`
- `sanitizeQuestion(q)`: `{ id, code, kind, content, multiSelect, options: q.options.map(o => ({ text: o.text })) }`
- 会话内按 id 加载题目**不加 scope 过滤**（会话合法性由 playerId+currentQuestionId 保证，题目范围已由 admin 校验器锁定），代码注释注明原因
